    ->     p_PrescriptionID INT

2. Function to Calculate Prescription Cost

MariaDB [pharmacy_db]> CREATE FUNCTION fn_calculate_prescription_cost(
    ->     p_PrescriptionID INT
    -> )
    -> RETURNS DECIMAL(10,2)
    -> DETERMINISTIC
    -> BEGIN
    ->     DECLARE v_TotalCost DECIMAL(10,2);
    ->
    ->     SELECT COALESCE(
    ->         SUM(pi.Quantity * m.UnitPrice),
    ->         0
    ->     )
    ->     INTO v_TotalCost
    ->     FROM PRESCRIPTION_ITEM pi
    ->     JOIN MEDICINE m
    ->         ON pi.MedicineID = m.MedicineID
    ->     WHERE pi.PrescriptionID = p_PrescriptionID;
    ->
    ->     RETURN v_TotalCost;
    -> END //
